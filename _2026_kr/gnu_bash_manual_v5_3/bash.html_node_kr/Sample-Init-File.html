8.3.3 예시 초기화 파일 ¶

다음은 inputrc 파일의 예시입니다. 이 예시는 키 바인딩, 변수 할당, 그리고 조건부 문법을 보여줍니다. 

# 이 파일은 GNU Readline 라이브러리를 사용하는 프로그램들의
# 줄 입력 편집 동작을 제어합니다. 해당 프로그램에는
# FTP, Bash, 그리고 GDB 등이 포함됩니다.
#
# C-x C-r 명령어로 inputrc 파일을 다시 읽어들일 수 있습니다.
# '#'으로 시작하는 줄은 주석입니다.
#
# 먼저, /etc/Inputrc에 있는 시스템 전역 바인딩 및
# 변수 할당을 포함시킵니다.
$include /etc/Inputrc

#
# emacs 모드를 위한 다양한 바인딩을 설정합니다.

set editing-mode emacs 

$if mode=emacs

Meta-Control-h:	backward-kill-word	# 함수 이름 뒤의 텍스트는 무시됩니다.

#
# 키패드(keypad) 모드에서의 화살표 키
#
#"\M-OD":        backward-char
#"\M-OC":        forward-char
#"\M-OA":        previous-history
#"\M-OB":        next-history
#
# ANSI 모드에서의 화살표 키
#
"\M-[D":        backward-char
"\M-[C":        forward-char
"\M-[A":        previous-history
"\M-[B":        next-history
#
# 8비트 키패드 모드에서의 화살표 키
#
#"\M-\C-OD":       backward-char
#"\M-\C-OC":       forward-char
#"\M-\C-OA":       previous-history
#"\M-\C-OB":       next-history
#
# 8비트 ANSI 모드에서의 화살표 키
#
#"\M-\C-[D":       backward-char
#"\M-\C-[C":       forward-char
#"\M-\C-[A":       previous-history
#"\M-\C-[B":       next-history

C-q: quoted-insert

$endif

# 예전 스타일의 바인딩. 공교롭게도 이것이 기본값입니다.
TAB: complete

# 쉘 상호작용에 편리한 매크로들
$if Bash
# path 편집
"\C-xp": "PATH=${PATH}\e\C-e\C-a\ef\C-f"
# 따옴표로 감싸진 단어를 입력할 준비 --
# 여는 따옴표와 닫는 큰따옴표를 삽입하고
# 여는 따옴표 바로 뒤로 커서 이동
"\C-x\"": "\"\"\C-b"
# 백슬래시 삽입 (시퀀스와 매크로에서의 
# 백슬래시 이스케이프 테스트)
"\C-x\\": "\\"
# 현재 또는 이전 단어를 따옴표로 감쌈
"\C-xq": "\eb\"\ef\""
# 바인딩되지 않은 줄 새로고침 기능을 위한 바인딩 추가
"\C-xr": redraw-current-line
# 현재 줄의 변수 편집
"\M-\C-v": "\C-a\C-k$\C-y\M-\C-e\C-a\y="
$endif

# 사용 가능한 경우 시각적 벨(visible bell) 사용
set bell-style visible

# 읽을 때 문자들을 7비트로 자르지 않음
set input-meta on

# iso-latin1 문자들을 메타 접두사 시퀀스로 변환하는 대신
# 그대로 삽입하도록 허용
set convert-meta off

# 여덟 번째 비트가 설정된 문자들을 메타 접두사 문자가 아닌
# 직접 표시함
set output-meta on

# 한 단어에 대해 가능한 완성 후보가 150개 이상인 경우,
# 모두 표시할지 여부를 사용자에게 물음
set completion-query-items 150

# FTP용 설정
$if Ftp
"\C-xg": "get \M-?"
"\C-xt": "put \M-?"
"\M-.": yank-last-arg
$endif
